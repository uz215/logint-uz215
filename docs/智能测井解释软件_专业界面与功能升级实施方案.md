### 2.3 线程模型（防冻结）

解释流程耗时长（4 万行数据 + OCR + 出图），**必须**放后台线程：

```python
# loggui/worker.py
from PyQt5.QtCore import QThread, pyqtSignal

class InterpretWorker(QThread):
    progress = pyqtSignal(str)      # 阶段文字
    finished = pyqtSignal(object)   # results dict
    failed   = pyqtSignal(str)

    def __init__(self, project_kwargs, mode='quicklook'):
        super().__init__()
        self.kw, self.mode = project_kwargs, mode

    def run(self):
        try:
            import logInt                      # 延迟导入，避免启动即加载 21k 行
            if self.mode == 'quicklook':
                r = logInt.quicklook_well(**self.kw)
            else:
                r = logInt.run_well_log_workflow(**self.kw)
            self.finished.emit(r)
        except Exception as e:
            import traceback
            self.failed.emit(traceback.format_exc())
```

⚠️ **注意 `logInt.py` 的模块级副作用**（已实测 @行 212-304）：导入即执行 `print_dependency_status()`、`setup_logging()`、设置 `plt.rcParams`。这是安全的（无阻塞、无 GUI），但会重设 matplotlib 后端参数——**必须在创建任何 Qt 画布之前完成导入**，否则 `Qt5Agg` 后端可能被覆盖。`app.py` 里按此顺序处理。

## 3. 阶段 0：地基修复（前置，约半天）

### 3.1 修复 `templates/qlpt.json`

**已验证的修复算法**（探针实测：修复 23 行后 JSON 完整解析，10 个顶层段全部恢复，`figure.dpi=150`、`curve_scale={'m':'1:200','ft':'1:240'}`、`legend.items=11`、`lithology.colors=34` 数值完好）：

损坏形态是"字符串闭合引号被 mojibake 产生的 `?` 吞掉"，两种：
- 值尾：`"note": "curve_scale 为...默认?  ,` ← 缺 `"`
- 键尾：`"娉ラ〉宀?: "#228B22",` ← 缺 `"`

修复脚本（状态机，已实测通过）：

```python
# scripts/repair_qlpt_template.py
"""无损修复 templates/qlpt.json 的语法结构。
只补回丢失的字符串闭合引号，不改动任何配置数值。
中文 mojibake 注释保持原样（不猜测原意），仅保证 JSON 可解析。
"""
import json, shutil, sys
from pathlib import Path

SRC = Path('templates/qlpt.json')
CLOSE_NEXT = set(':,}]')


def repair_line(raw: str) -> str:
    out, in_str, i, n = [], False, 0, len(raw)
    while i < n:
        c = raw[i]
        if in_str:
            if c == '\\':
                out.append(raw[i:i + 2]); i += 2; continue
            if c == '"':
                in_str = False; out.append(c); i += 1; continue
            # 私有区字符(mojibake残留)或 ?，其后紧跟 : , } ] 或行尾 → 此处丢了闭合引号
            if c == '?' or 0xE000 <= ord(c) <= 0xF8FF:
                j = i + 1
                while j < n and raw[j] in ' \t':
                    j += 1
                if j >= n or raw[j] in CLOSE_NEXT:
                    out.append(c); out.append('"'); in_str = False; i += 1; continue
            out.append(c); i += 1
        else:
            if c == '"':
                in_str = True
            out.append(c); i += 1
    return ''.join(out)


def main():
    text = SRC.read_text(encoding='utf-8')
    lines = text.split('\n')
    fixed = [repair_line(L.rstrip('\r')) + ('\r' if L.endswith('\r') else '')
             for L in lines]
    out = '\n'.join(fixed)

    d = json.loads(out)                     # 解析失败会抛异常，不会写文件

    bak = SRC.with_suffix('.json.corrupt_bak')
    if not bak.exists():
        shutil.copy2(SRC, bak)              # 保留损坏原件，可回溯
    SRC.write_text(out, encoding='utf-8')

    print('修复完成，顶层段:', list(d.keys()))
    print('figure.dpi =', d['figure']['dpi'])
    print('legend.items =', len(d['legend']['items']))
    print('lithology.colors =', len(d['lithology']['colors']))
    print('原损坏文件已备份到', bak)


if __name__ == '__main__':
    sys.exit(main())
```

**执行前需你确认**：这会修改现有配置文件（自动留 `.corrupt_bak` 备份）。

**关于中文注释**：mojibake 是 UTF-8 被当 GBK 解码所致。`gb18030` 反解可部分还原（实测 `快速解释图 Quick Look Plot（单页转置版式，默认`），但**不可逆字符已丢失**（私有区 `\ue1be` 等无法还原）。方案：语法先修复保证程序可用，中文注释在阶段 2 的模板编辑器里**由你重新录入**——那时你正在图形界面里配置道，顺手写清每段 note 比现在猜编码更可靠。

### 3.3 打通模板读取通道（地基中的地基，约 1 天）

> **定位澄清**：`_q_layout` / `_layout` 硬编码是当初的**应急措施**——为了让 quicklook 先有一个固定模板能出图。它不是需要保护的资产，而是需要被替换的临时实现。**模板才是布局的唯一源头（single source of truth）**，代码里的硬编码只保留为"模板缺失时的兜底"。
>
> 这一步必须在阶段 1 之前完成。否则会出现最坏的情况：交互画布读模板、静态出图读硬编码，**两套布局并存且互相分叉**——而这正是当前模板数值与代码数值已经不一致的成因。

#### 3.3.1 统一布局解析函数

新建 `loggui/track_spec.py` 中的解析器，**同时供静态出图和交互画布调用**，杜绝两套逻辑：

```python
# loggui/track_spec.py 追加

# 道 ID → 显示条件。None 表示无条件显示；字符串键表示依赖该运行时标志
TRACK_CONDITIONS = {
    'RES_TRAD':  'has_res_trad',
    'RES_ARRAY': 'has_res_array',
    'GAS':       'has_gas',
}

# 硬编码兜底布局：与 logvis.py:4526-4563 / 4570-4602 当前值逐一对齐。
# 仅当模板缺失 track.layout 时使用，保证老模板出图结果完全不变。
FALLBACK_LAYOUT = {
    'qlpt': [
        ('LITHO', 1.2), ('RES_TRAD', 1.4), ('RES_ARRAY', 1.4),
        ('POR', 1.0), ('DEPTH', 0.30), ('SW', 0.6), ('POR2', 0.6),
        ('GAS', 'gas'), ('PROFILE', 1.0), ('INTERP', 0.42),
    ],
    'cmpt': [
        ('LITHO', 1.0), ('RES_TRAD', 1.4), ('RES_ARRAY', 1.4),
        ('POR', 1.25), ('DEPTH', 0.5), ('SWPOR', 1.0),
        ('GAS', 'gas'), ('LITHO_FILL', 1.0), ('FACIES', 1.0),
        ('PROFILE', 1.0), ('INTERP', 1.0),
    ],
}


def resolve_track_layout(tpl, kind, flags, gas_track_width=1.0):
    """解析出道布局：[(道ID, 宽度float), ...]

    参数
    ----
    tpl   : logvis.load_plot_template(kind) 的返回（可为 None）
    kind  : 'qlpt'（快速解释）| 'cmpt'（综合图）
    flags : {'has_res_trad': bool, 'has_res_array': bool, 'has_gas': bool}
            运行时可用道判定，对应代码里的 trad_res_cols / array_res_cols / has_gas
    gas_track_width : 气测道实算宽度，用于替换模板里的 "gas" 占位

    优先级
    ------
    模板 track.layout（有效项）→ FALLBACK_LAYOUT[kind]（应急兜底）

    约定（与 cmpt.json 现有 widths_with_gas 的 "gas" 占位写法一致）
    --------------------------------------------------------------
    width 可为数字，或字符串 "gas"（表示用 gas_track_width 实算值）
    enabled 可为 true/false，或 "auto"（表示按 flags 判定，缺省即 auto）
    """
    cfg = ((tpl or {}).get('track') or {}).get('layout')
    raw = cfg if isinstance(cfg, list) and cfg else FALLBACK_LAYOUT.get(kind, [])
    from_template = cfg is not None

    out = []
    for it in raw:
        if isinstance(it, dict):
            tid, width, enabled = it.get('id'), it.get('width', 1.0), it.get('enabled', 'auto')
        else:                       # 兜底布局是 (id, width) 元组
            tid, width, enabled = it[0], it[1], 'auto'
        if not tid:
            continue
        # enabled: false → 显式关闭；true → 强制开启；auto → 按运行时标志判定
        if enabled is False:
            continue
        if enabled == 'auto':
            cond = TRACK_CONDITIONS.get(tid)
            if cond and not flags.get(cond):
                continue
        if width == 'gas':
            if not flags.get('has_gas'):
                continue
            width = gas_track_width
        out.append((tid, float(width)))

    if not out:                     # 模板写了但全被禁用 → 回落兜底
        return resolve_track_layout(None, kind, flags, gas_track_width), False
    return out, from_template
```

**关键设计点**：
- `enabled: "auto"` 沿用你项目里 `reg_tracks.enabled = "auto"` 的既有约定（实测在 `qlpt.json` 中），不另造术语
- `width: "gas"` 沿用 `cmpt.json` 的 `widths_with_gas` 里 `"gas"` 占位写法（实测存在），由 `gas_track_width` 实算替换
- 返回值带 `from_template` 布尔标志，**出图时打日志说明布局来源**，避免以后又出现"改了模板没反应还不知道为什么"
- 兜底布局数值**逐一对齐当前代码硬编码**，保证老模板出图像素级不变

### 4.1 性能地基：视窗裁剪 + 包络降采样

**为什么不能只做均值抽稀**：测井曲线的尖峰（薄层、扩径、煤层响应）是关键地质信息，均值抽稀会抹平。必须取 **min/max 包络**。

```python
# loggui/decimate.py
"""深度视窗内的包络降采样。
目标：把 4 万行数据压到 ~2×画布像素高度的点数，且保留尖峰。
"""
import numpy as np


def envelope_decimate(values, n_bins):
    """返回 (抽稀后的值数组, 对应的原始索引数组)。

    每个 bin 取 min 和 max 两点，并按原始先后顺序排列，
    保证连线形状忠实于原曲线的包络。
    全 NaN 的 bin 跳过（不产生伪值）。
    """
    v = np.asarray(values, dtype=float)
    n = v.size
    if n_bins <= 0 or n <= n_bins * 2:
        idx = np.arange(n)
        return v, idx

    edges = np.linspace(0, n, n_bins + 1).astype(int)
    out_v, out_i = [], []
    for a, b in zip(edges[:-1], edges[1:]):
        if b <= a:
            continue
        seg = v[a:b]
        if np.all(np.isnan(seg)):
            continue
        with np.errstate(invalid='ignore'):
            imin = int(np.nanargmin(seg))
            imax = int(np.nanargmax(seg))
        for k in sorted({imin, imax}):
            out_v.append(seg[k])
            out_i.append(a + k)
    if not out_v:
        idx = np.arange(n)
        return v, idx
    return np.asarray(out_v), np.asarray(out_i)


def depth_window_slice(depth, top, bottom):
    """返回深度落在 [top, bottom] 内的布尔掩码（深度轴假定单调）。"""
    d = np.asarray(depth, dtype=float)
    return (d >= top) & (d <= bottom)
```

### 4.2 道配置数据类

```python
# loggui/track_spec.py
"""道与曲线的配置数据结构。
阶段1 用于交互画布；阶段2 的模板编辑器读写同一结构 → 天然打通。
字段命名对齐 templates/*.json 现有风格（英文小写下划线）。
"""
from dataclasses import dataclass, field, asdict


@dataclass
class CurveSpec:
    column: str                      # DataFrame 列名，如 'GR' / 'RT' / 'Sw'
    label: str = ''                  # 道头显示名
    color: str = '#1f77b4'
    linestyle: str = '-'
    linewidth: float = 0.8
    scale: str = 'lin'               # 'lin' | 'log'
    xmin: float | None = None        # None = 自动（百分位）
    xmax: float | None = None
    fill_to: float | None = None     # 填充到该 x 值（如 Vsh 填到 0）
    fill_color: str | None = None


@dataclass
class TrackSpec:
    id: str                          # 'LITHO' / 'RES_TRAD' / 'POR' / 'DEPTH' / 'SW' / 'INTERP' ...
    name: str                        # 中文道名，道头显示
    kind: str = 'curve'              # curve | lithology | depth | interpretation | profile | gas | facies
    width: float = 1.0
    curves: list[CurveSpec] = field(default_factory=list)
    show_grid: bool = True
    header_lines: int = 3            # 道头占几行文字高度
    # 启停：True/False 显式开关，'auto' 按运行时标志判定（沿用模板 reg_tracks 既有约定）
    # 仅编辑器使用；出图侧由 resolve_track_layout 消费，解析后不再需要
    enabled: bool | str = 'auto'

    def to_dict(self):
        return asdict(self)

    @classmethod
    def from_dict(cls, d):
        cs = [CurveSpec(**c) for c in d.get('curves', [])]
        known = {f for f in cls.__dataclass_fields__}
        return cls(curves=cs, **{k: v for k, v in d.items() if k in known and k != 'curves'})


# 道 ID → 绘制类型与默认曲线。只描述"道怎么画"，不描述"有哪些道、多宽"
# —— 后者由 resolve_track_layout 从模板解析，避免与静态出图分叉。
TRACK_KINDS = {
    'LITHO':      ('lithology',      []),
    'RES_TRAD':   ('curve',          ['RT', 'LLD', 'MSFL']),
    'RES_ARRAY':  ('curve',          ['AT10', 'AT20', 'AT30', 'AT60', 'AT90']),
    'POR':        ('curve',          ['DEN', 'NPHI', 'AC']),
    'DEPTH':      ('depth',          []),
    'SW':         ('curve',          ['Sw']),
    'SWPOR':      ('curve',          ['Sw', 'porosity']),
    'POR2':       ('curve',          ['porosity']),
    'GAS':        ('gas',            ['TG']),
    'PROFILE':    ('profile',        []),
    'LITHO_FILL': ('lithology',      []),
    'FACIES':     ('facies',         []),
    'INTERP':     ('interpretation', []),
}


def build_tracks(tpl, kind, flags, gas_track_width=1.0, available_columns=None):
    """从模板解析布局 → 组装成 TrackSpec 列表。

    与 logvis 静态出图调用同一个 resolve_track_layout，
    因此交互画布与甲方交付图的道顺序、道宽、启停状态永远一致。
    """
    layout, _from_tpl = resolve_track_layout(tpl, kind, flags, gas_track_width)
    defs = ((tpl or {}).get('track') or {}).get('track_defs') or {}
    names = {t.get('id'): t.get('name') for t in
             (((tpl or {}).get('track') or {}).get('layout') or []) if isinstance(t, dict)}
    cols = set(available_columns or [])
    out = []
    for tid, width in layout:
        d = defs.get(tid, {})
        tk, default_curves = TRACK_KINDS.get(tid, ('curve', []))
        curves = [CurveSpec(**c) for c in d.get('curves', [])]
        if not curves:
            # 模板未定义曲线 → 按道 ID 的默认候选列，取数据中实际存在的
            curves = [CurveSpec(column=c) for c in default_curves
                      if not cols or c in cols]
        out.append(TrackSpec(
            id=tid, name=names.get(tid) or d.get('name') or tid,
            kind=d.get('kind', tk), width=width, curves=curves,
            show_grid=d.get('show_grid', True)))
    return out


# 兼容旧引用名：仅在模板完全不可用时作为最后兜底
def default_quicklook_tracks(available_columns=None):
    return build_tracks(None, 'qlpt',
                        {'has_res_trad': True, 'has_res_array': False, 'has_gas': False},
                        available_columns=available_columns)
```

⚠️ **不要在这里再写一份宽度常量**。原本草案里有个 `DEFAULT_QUICKLOOK_TRACKS` 硬编码列表，那会重演当前的问题——模板改了、界面没变。宽度值统一来自 `FALLBACK_LAYOUT`（§3.3.1，逐一对齐 `logvis.py:4526-4563` / `4570-4602` 实测硬编码），保证交互画布与甲方静态图**版式一致**，且只有一个地方能改它。

### 4.3 交互画布主体

```python
# loggui/track_canvas.py
"""多道交互测井图画布。
交互：滚轮=深度缩放，左键拖拽=平移，右键拖拽=深度框选放大，
     移动鼠标=跨道深度游标 + 实时读值，双击=复位全井。
所有道共享深度轴（sharey），保证严格深度对齐。
"""
import numpy as np
import matplotlib
matplotlib.use('Qt5Agg')
from matplotlib.figure import Figure
from matplotlib.backends.backend_qt5agg import FigureCanvasQTAgg as FigureCanvas
from PyQt5.QtCore import pyqtSignal, Qt

from loggui.decimate import envelope_decimate, depth_window_slice
from loggui.track_spec import TrackSpec

HEADER_FRAC = 0.10          # 道头占画布高度比例
MIN_SPAN = 2.0              # 最小可视深度跨度（单位同深度轴），防止无限放大
CURSOR_COLOR = '#e74c3c'


class TrackCanvas(FigureCanvas):
    cursorMoved = pyqtSignal(float, dict)     # (深度, {列名: 值})
    rangeChanged = pyqtSignal(float, float)   # (top, bottom)

    def __init__(self, parent=None):
        self.fig = Figure(facecolor='white', tight_layout=False)
        super().__init__(self.fig)
        self.setParent(parent)
        self.project = None
        self.tracks: list[TrackSpec] = []
        self.axes = []
        self.top, self.bottom = 0.0, 100.0
        self.full_top, self.full_bottom = 0.0, 100.0
        self._cursor_lines = []
        self._drag = None
        self.setMouseTracking(True)
        self._connect()

    # ---------- 数据装载 ----------
    def set_project(self, project, tracks=None):
        """project: WellProject；tracks: 道配置（None 用默认布局）"""
        self.project = project
        self.tracks = tracks or []
        d = project.depth
        self.full_top, self.full_bottom = float(d[0]), float(d[-1])
        self.top, self.bottom = self.full_top, self.full_bottom
        self._build_axes()
        self.refresh()

    def _build_axes(self):
        """按 TrackSpec.width 比例横向排布，全部 sharey 到第一个道。"""
        self.fig.clear()
        self.axes = []
        widths = [max(t.width, 0.05) for t in self.tracks]
        total = sum(widths)
        x = 0.0
        for t, w in zip(self.tracks, widths):
            frac = w / total
            ax = self.fig.add_axes([x, 0.02, frac, 1.0 - HEADER_FRAC - 0.04])
            if self.axes:
                ax.sharey(self.axes[0])           # 关键：深度严格对齐
            ax.set_xticks([])
            self.axes.append(ax)
            x += frac
        self._draw_headers()

    def _draw_headers(self):
        """道头：道名 + 曲线名/量程，用 fig 坐标绘制，不占数据区。"""
        for ax, t in zip(self.axes, self.tracks):
            pos = ax.get_position()
            self.fig.text(pos.x0 + pos.width / 2, pos.y1 + 0.045, t.name,
                          ha='center', va='center', fontsize=9, weight='bold')
            sub = [c.label or c.column for c in t.curves]
            if sub:
                self.fig.text(pos.x0 + pos.width / 2, pos.y1 + 0.015,
                              ' / '.join(sub), ha='center', va='center',
                              fontsize=7, color='#555555')

    # ---------- 绘制 ----------
    def refresh(self):
        """重绘当前深度窗口内的所有道（含降采样）。"""
        if self.project is None or not self.axes:
            return
        df = self.project.processed
        depth = self.project.depth
        mask = depth_window_slice(depth, self.top, self.bottom)
        d_vis = depth[mask]
        if d_vis.size < 2:
            self.draw_idle()
            return
        # 降采样目标点数：与画布像素高度挂钩，兼顾清晰度与性能
        px_h = max(self.height() - int(self.height() * HEADER_FRAC), 100)
        n_bins = int(px_h * 0.75)

        for ax, t in zip(self.axes, self.tracks):
            ax.clear()
            ax.set_xticks([])
            if t.kind == 'depth':
                self._draw_depth_track(ax, d_vis)
            elif t.kind == 'lithology':
                self._draw_lithology_track(ax, d_vis, mask)
            elif t.kind == 'interpretation':
                self._draw_interp_track(ax, d_vis, mask)
            else:
                self._draw_curve_track(ax, t, df, d_vis, mask, n_bins)
            ax.set_ylim(self.bottom, self.top)     # 深度向下递增 → 反向
        self._reapply_cursor()
        self.draw_idle()
        self.rangeChanged.emit(self.top, self.bottom)

    def _draw_curve_track(self, ax, t, df, d_vis, mask, n_bins):
        cols = [c.column for c in t.curves if c.column in df.columns]
        if not cols:
            ax.text(0.5, 0.5, '无数据', transform=ax.transAxes,
                    ha='center', va='center', fontsize=8, color='#999')
            return
        for c in t.curves:
            if c.column not in df.columns:
                continue
            raw = df.loc[mask, c.column].to_numpy(dtype=float)
            v, idx = envelope_decimate(raw, n_bins)
            dv = d_vis[idx] if idx.size else d_vis[:0]
            if c.fill_to is not None:
                ax.fill_betweenx(dv, c.fill_to, v, color=c.fill_color or c.color,
                                 alpha=0.25, linewidth=0)
            ax.plot(v, dv, color=c.color, ls=c.linestyle, lw=c.linewidth)
        # x 轴量程：显式优先，否则用 2-98 百分位（抗异常值）
        lo = [c.xmin for c in t.curves if c.xmin is not None]
        hi = [c.xmax for c in t.curves if c.xmax is not None]
        if lo and hi:
            ax.set_xlim(min(lo), max(hi))
        else:
            vals = np.concatenate([df.loc[mask, c].dropna().to_numpy(dtype=float)
                                   for c in cols]) if cols else np.array([])
            if vals.size:
                a, b = np.nanpercentile(vals, [2, 98])
                pad = (b - a) * 0.08 or 1.0
                ax.set_xlim(a - pad, b + pad)
        if any(c.scale == 'log' for c in t.curves):
            ax.set_xscale('log')
        if t.show_grid:
            ax.grid(axis='x', ls=':', lw=0.4, color='#cccccc', alpha=0.7)
        ax.tick_params(labelbottom=False, length=0)

    def _draw_depth_track(self, ax, d_vis):
        """深度道：主/次刻度。步长随窗口跨度自适应。"""
        span = abs(self.bottom - self.top)
        unit = self.project.depth_unit or 'm'
        for target in (1000, 500, 200, 100, 50, 20, 10, 5, 2, 1):
            if span / target >= 4:
                step = target
                break
        else:
            step = 0.5
        start = np.ceil(min(self.top, self.bottom) / step) * step
        ticks = np.arange(start, max(self.top, self.bottom) + step, step)
        ax.set_xlim(0, 1)
        ax.set_xticks([])
        for tv in ticks:
            ax.axhline(tv, color='#888888', lw=0.5)
            ax.text(0.5, tv, f'{tv:g}', ha='center', va='center',
                    fontsize=6.5, color='#333333',
                    bbox=dict(fc='white', ec='none', pad=0.6))
        ax.set_title(f'深度({unit})', fontsize=7, pad=1)

    def _draw_lithology_track(self, ax, d_vis, mask):
        """岩性道：复用 logvis 已有的图案绘制，保证与静态出图一致。"""
        df = self.project.processed
        col = next((c for c in ('lithology_identification', 'lithology_array',
                                'lithology') if c in df.columns), None)
        if col is None:
            ax.text(0.5, 0.5, '无岩性', transform=ax.transAxes, ha='center',
                    va='center', fontsize=8, color='#999')
            return
        ax.set_xlim(0, 1)
        ax.set_xticks([])
        lith = df.loc[mask, col].astype(str)
        dd = d_vis
        # 分段：连续同岩性合并为一段，再调 logvis 的图案函数
        try:
            import logvis
            runs, cur = [], None
            for i, name in enumerate(lith.to_numpy()):
                if cur is None or name != cur[2]:
                    if cur:
                        runs.append(cur)
                    cur = [dd[i], dd[i], name]
                else:
                    cur[1] = dd[i]
            if cur:
                runs.append(cur)
            for z0, z1, name in runs:
                logvis.draw_lithology_pattern(ax, name, z0, z1, x0=0.0, x1=1.0)
        except Exception:
            # 兜底：色块表示，不让画布因单个岩性名失败而空白
            from logvis import _tpl_colors if False else None  # noqa
            for z0, z1, name in runs:
                ax.axhspan(z0, z1, color='#dddddd', lw=0)
                ax.text(0.5, (z0 + z1) / 2, name, ha='center', va='center', fontsize=6)

    def _draw_interp_track(self, ax, d_vis, mask):
        """解释结论道：按流体类别着色（色值取自 templates/cppt.json fluid.colors）。"""
        df = self.project.processed
        if 'fluid_identification' not in df.columns:
            ax.set_xticks([]); return
        ax.set_xlim(0, 1); ax.set_xticks([])
        colors = self._fluid_colors()
        vals = df.loc[mask, 'fluid_identification'].astype(str)
        for v, dv in zip(vals.to_numpy(), d_vis):
            ax.axhspan(dv, dv, color=colors.get(v.upper(), '#bdbdbd'), lw=1.2)

    def _fluid_colors(self):
        try:
            import logvis
            tpl = logvis.load_plot_template('cppt') or {}
            return tpl.get('fluid', {}).get('colors', {})
        except Exception:
            return {}

    # ---------- 交互 ----------
    def _connect(self):
        self.mpl_connect('scroll_event', self._on_scroll)
        self.mpl_connect('button_press_event', self._on_press)
        self.mpl_connect('motion_notify_event', self._on_move)
        self.mpl_connect('button_release_event', self._on_release)

    def _ydata_at(self, event):
        if not self.axes or event.inaxes is None:
            return None
        return event.ydata

    def _on_scroll(self, event):
        """滚轮：以鼠标所在深度为锚点缩放。"""
        if event.ydata is None:
            return
        anchor = float(event.ydata)
        factor = 0.8 if event.button == 'up' else 1.25
        span = (self.bottom - self.top) * factor
        if span < MIN_SPAN:
            span = MIN_SPAN
        if span > (self.full_bottom - self.full_top):
            span = self.full_bottom - self.full_top
        ratio = (anchor - self.top) / max(self.bottom - self.top, 1e-9)
        self.top = anchor - ratio * span
        self.bottom = self.top + span
        self._clamp()
        self.refresh()

    def _on_press(self, event):
        if event.ydata is None:
            return
        if event.button == 1:
            self._drag = ('pan', float(event.ydata), self.top, self.bottom)
        elif event.button == 3:
            self._drag = ('zoom', float(event.ydata), None, None)
        elif event.button == 2:
            self.top, self.bottom = self.full_top, self.full_bottom
            self.refresh()

    def _on_move(self, event):
        if event.ydata is None:
            return
        y = float(event.ydata)
        if self._drag and self._drag[0] == 'pan':
            _, y0, t0, b0 = self._drag
            dy = y0 - y
            self.top, self.bottom = t0 + dy, b0 + dy
            self._clamp()
            self.refresh()
            return
        self._update_cursor(y)

    def _on_release(self, event):
        if self._drag and self._drag[0] == 'zoom' and event.ydata is not None:
            y0 = self._drag[1]
            y1 = float(event.ydata)
            if abs(y1 - y0) > MIN_SPAN / 4:
                self.top, self.bottom = min(y0, y1), max(y0, y1)
                self._clamp()
                self.refresh()
        self._drag = None

    def _clamp(self):
        span = self.bottom - self.top
        if self.top < self.full_top:
            self.top, self.bottom = self.full_top, self.full_top + span
        if self.bottom > self.full_bottom:
            self.bottom = self.full_bottom
            self.top = self.bottom - span

    def _update_cursor(self, y):
        """跨道游标 + 读值。用 blit 只重绘游标层，避免全图重绘。"""
        for ln in self._cursor_lines:
            ln.remove()
        self._cursor_lines = []
        for ax in self.axes:
            ln = ax.axhline(y, color=CURSOR_COLOR, lw=0.9, ls='--')
            self._cursor_lines.append(ln)
        self.draw_idle()
        self.cursorMoved.emit(y, self._read_values(y))

    def _reapply_cursor(self):
        self._cursor_lines = []

    def _read_values(self, y):
        """最近邻取该深度所有曲线值。"""
        if self.project is None:
            return {}
        d = self.project.depth
        i = int(np.nanargmin(np.abs(d - y)))
        row = self.project.processed.iloc[i]
        out = {'DEPTH': float(d[i])}
        out.update({k: (None if _isnan(v) else float(v))
                    for k, v in row.items() if _isnum(v)})
        return out


def _isnum(v):
    try:
        float(v); return True
    except Exception:
        return False


def _isnan(v):
    try:
        return float(v) != float(v)
    except Exception:
        return True
```

### 5.1 本阶段的真实难点：双向一致性

编辑器要解决的不再是"能不能读到模板"，而是三个一致性问题：

| 难点 | 说明 | 对策 |
|---|---|---|
| **预览与出图一致** | 编辑器中栏预览用 `TrackCanvas`，最终交付图用 `logvis` 静态出图。两者必须同源 | 双方都调 `resolve_track_layout`（§3.3.1），预览即真相 |

```python
        for t in self.tracks:
            self.list.addItem(QListWidgetItem(t.name))
        self._sync_list_labels()
        self._refresh_preview()
```

```python
# loggui/crossplot_widget.py
"""交互式交会图 + 参数标定。
交互：
  矩形框选 / 套索选择 → 选中点集高亮 + 统计面板（均值/中位/分位/点数）
  按流体类别或岩性着色（色值取自 cppt.json）
  框选后一键回归 → 显示拟合线/方程/R²
  框选点集 → 推送到标定面板 → 调 logsys.calibrate → 结果写回 cutoff
  深度联动：点击交会图上某点 → 测井图画布跳到该深度并置游标
"""
import numpy as np
import matplotlib
matplotlib.use('Qt5Agg')
from matplotlib.figure import Figure
from matplotlib.backends.backend_qt5agg import FigureCanvasQTAgg as FigureCanvas
from matplotlib.widgets import RectangleSelector, LassoSelector
from PyQt5.QtCore import pyqtSignal


def resolve_column(df, candidates):
    """按 cppt.json 约定：候选列名按顺序取第一个存在的。"""
    for c in candidates or []:
        if c in df.columns:
            return c
    return None


class CrossplotCanvas(FigureCanvas):
    pointPicked = pyqtSignal(float)        # 深度 → 联动测井图画布
    selectionMade = pyqtSignal(object)     # 选中点索引数组 → 标定面板

    def __init__(self, parent=None):
        self.fig = Figure(facecolor='white')
        super().__init__(self.fig)
        self.ax = self.fig.add_subplot(111)
        self.df = None
        self.xcol = self.ycol = None
        self.depth = None
        self.x = self.y = None
        self.sel_mask = np.zeros(0, dtype=bool)
        self.sc = None
        self._selector = None
        self._reg_artist = None
        self.mpl_connect('button_press_event', self._on_click)

    def load_spec(self, df, depth, spec: dict, color_by='fluid_identification',
                  colors=None):
        """spec: cppt.json crossplots 中的一项。"""
        self.df, self.depth = df, np.asarray(depth, float)
        self.xcol = resolve_column(df, spec.get('x_curves'))
        self.ycol = resolve_column(df, spec.get('y_curves'))
        if not self.xcol or not self.ycol:
            self.ax.clear()
            self.ax.text(0.5, 0.5, '缺少曲线：%s / %s' % (
                self.xcol or spec.get('x_curves'), self.ycol or spec.get('y_curves')),
                transform=self.ax.transAxes, ha='center', va='center', color='#c00')
            self.draw_idle()
            return
        self.x = df[self.xcol].to_numpy(dtype=float)
        self.y = df[self.ycol].to_numpy(dtype=float)
        ok = np.isfinite(self.x) & np.isfinite(self.y)
        self.sel_mask = np.zeros(self.x.size, dtype=bool)

        self.ax.clear()
        cmap_colors = colors or {}
        if color_by in df.columns and cmap_colors:
            cats = df[color_by].astype(str).str.upper().to_numpy()
            for cat in np.unique(cats):
                m = ok & (cats == cat)
                if m.any():
                    self.ax.scatter(self.x[m], self.y[m], s=5, alpha=0.55,
                                    color=cmap_colors.get(cat, '#999999'),
                                    label=cat, linewidths=0)
            self.ax.legend(loc='best', fontsize=7, framealpha=0.8, markerscale=2)
        else:
            self.ax.scatter(self.x[ok], self.y[ok], s=5, alpha=0.5,
                            color='#2471a3', linewidths=0)

        if spec.get('x_log'):
            self.ax.set_xscale('log')
        self.ax.set_xlabel(spec.get('x_label') or self.xcol, fontsize=9)
        self.ax.set_ylabel(spec.get('y_label') or self.ycol, fontsize=9)
        self.ax.set_title(spec.get('purpose', ''), fontsize=10)
        self.ax.grid(ls=':', lw=0.4, color='#cccccc', alpha=0.7)
        self._enable_rectangle()
        self.draw_idle()

    def _enable_rectangle(self):
        self._selector = RectangleSelector(
            self.ax, self._on_rect, useblit=True, button=[1],
            minspanx=1e-9, minspany=1e-9, spancoords='data', interactive=False)

    def _on_rect(self, eclick, erelease):
        x0, x1 = sorted((eclick.xdata, erelease.xdata))
        y0, y1 = sorted((eclick.ydata, erelease.ydata))
        m = (self.x >= x0) & (self.x <= x1) & (self.y >= y0) & (self.y <= y1)
        m &= np.isfinite(self.x) & np.isfinite(self.y)
        self.sel_mask = m
        self._highlight()
        self.selectionMade.emit(np.flatnonzero(m))

    def _highlight(self):
        for a in getattr(self, '_hl', []):
            a.remove()
        self._hl = []
        if self.sel_mask.any():
            art = self.ax.scatter(self.x[self.sel_mask], self.y[self.sel_mask],
                                  s=26, facecolors='none', edgecolors='#e74c3c',
                                  linewidths=0.9)
            self._hl.append(art)
        self.draw_idle()

    def stats(self):
        if not self.sel_mask.any():
            return None
        xs, ys = self.x[self.sel_mask], self.y[self.sel_mask]
        return {'n': int(self.sel_mask.sum()),
                'x_mean': float(np.mean(xs)), 'x_med': float(np.median(xs)),
                'x_p10': float(np.percentile(xs, 10)), 'x_p90': float(np.percentile(xs, 90)),
                'y_mean': float(np.mean(ys)), 'y_med': float(np.median(ys)),
                'y_p10': float(np.percentile(ys, 10)), 'y_p90': float(np.percentile(ys, 90))}

    def regress(self, degree=1):
        """对选中点集做多项式回归，返回 (系数, R²)；对数轴先取 log。"""
        if self.sel_mask.sum() < degree + 2:
            return None, None
        xs, ys = self.x[self.sel_mask], self.y[self.sel_mask]
        if self.ax.get_xscale() == 'log':
            xs = np.log10(np.clip(xs, 1e-9, None))
        coef = np.polyfit(xs, ys, degree)
        pred = np.polyval(coef, xs)
        ss_res = float(np.sum((ys - pred) ** 2))
        ss_tot = float(np.sum((ys - np.mean(ys)) ** 2))
        r2 = 1 - ss_res / ss_tot if ss_tot > 0 else float('nan')
        self.draw_regression(coef, degree)
        return coef, r2

    def draw_regression(self, coef, degree):
        if self._reg_artist:
            self._reg_artist.remove()
        xs = np.linspace(np.nanmin(self.x[self.sel_mask]),
                         np.nanmax(self.x[self.sel_mask]), 200)
        px = np.log10(np.clip(xs, 1e-9, None)) if self.ax.get_xscale() == 'log' else xs
        ys = np.polyval(coef, px)
        self._reg_artist, = self.ax.plot(xs, ys, color='#8e44ad', lw=1.8, ls='--')
        self.draw_idle()

    def _on_click(self, event):
        """单击最近点 → 发深度，联动测井图。"""
        if event.inaxes is not self.ax or event.button != 1 or self.x is None:
            return
        if self._selector and self._selector.active:
            return
        d = np.hypot((self.x - event.xdata) / (np.ptp(self.x) + 1e-9),
                     (self.y - event.ydata) / (np.ptp(self.y) + 1e-9))
        d[~np.isfinite(self.x) | ~np.isfinite(self.y)] = np.inf
        i = int(np.nanargmin(d))
        if np.isfinite(d[i]) and d[i] < 0.05:
            self.pointPicked.emit(float(self.depth[i]))
```

## 7. 阶段 4：渗透率与束缚水算法补强（约 2 天）

**实测缺口**：全库检索 Timur / Coates / SDR / Swirr / 束缚水 → **0 处实现**（仅有核磁渗透率曲线的**读取语义** @`logInt.py:2275`）。这是功能完整度上最实的短板。

全部为公开文献模型，自主实现：

```python
# loggui/petrophysics.py
"""渗透率与束缚水饱和度模型（公开文献公式，自主实现）。

参考文献：
  Archie (1942)         电阻率指数与含水饱和度关系
  Timur (1968)          基于束缚水与孔隙度的渗透率经验式
  Coates & Denoo (1974) 核磁自由流体/束缚流体比渗透率
  Coates et al. (1991)  SDR（Schlumberger-Doll Research）核磁渗透率
  Wyllie-Rose           束缚水饱和度经验式
所有模型均需标定系数，默认值为文献常用初值，应经本区块岩心标定后使用。
"""
import numpy as np


defarchie_sw(rt, rw, a=1.0, m=2.0, phi=None):
    """Archie 含水饱和度。rt/rw 单位需一致；phi 为小数（非百分数）。"""
    rt = np.asarray(rt, float); phi = np.asarray(phi, float)
    with np.errstate(divide='ignore', invalid='ignore'):
        sw = ((a * rw) / (rt * np.power(phi, m))) ** 0.5
    return np.clip(sw, 0.0, 1.0)


def timur_perm(phi, swirr, a=100.0):
    """Timur (1968)：K = (a * phi^2 / Swirr)^2，phi 小数，K 单位 mD。
    适用砂岩；a 默认 100（文献常用），需用本区岩心标定。
    """
    phi = np.asarray(phi, float); swirr = np.asarray(swirr, float)
    with np.errstate(divide='ignore', invalid='ignore'):
        k = (a * np.power(phi, 2.0) / swirr) ** 2
    return np.where(np.isfinite(k), k, np.nan)


def coates_perm(phi, ffv, bfv, c=0.0135):
    """Coates & Denoo (1974) 核磁：K = (phi/c)^4 * (FFV/BVW)^2，K 单位 mD。
    phi 小数，ffv/bfv 为自由/束缚流体体积分数。
    """
    phi = np.asarray(phi, float)
    with np.errstate(divide='ignore', invalid='ignore'):
        ratio = np.where(bfv > 1e-6, ffv / np.maximum(bfv, 1e-6), np.nan)
        k = np.power(phi / c, 4.0) * np.power(ratio, 2.0)
    return np.where(np.isfinite(k), k, np.nan)


def sdr_perm(phi, t2lm, a=4.0, c=0.01):
    """SDR (Coates et al. 1991)：K = (phi/c)^4 * (T2LM)^2，K 单位 mD。
    phi 小数，t2lm 为 T2 几何均值（ms）。a 为孔隙度指数（常取 4）。
    """
    phi = np.asarray(phi, float)
    with np.errstate(divide='ignore', invalid='ignore'):
        k = np.power(phi / c, a) * np.power(t2lm, 2.0)
    return np.where(np.isfinite(k), k, np.nan)


def wyllie_rose_swirr(phi, k, c=250.0):
    """Wyllie-Rose 束缚水饱和度：Swirr = sqrt(c * phi / K)（phi 百分数，K 单位 mD）。
    仅适用于水湿岩石；c 需岩心标定。
    """
    phi = np.asarray(phi, float); k = np.asarray(k, float)
    with np.errstate(divide='ignore', invalid='ignore'):
        swirr = np.sqrt(c * phi / np.maximum(k, 1e-9))
    return np.clip(swirr, 0.0, 1.0)


def add_perm_columns(df, phi_col='porosity', swirr_col='Swirr',
                     prefix='perm_'):
    """把渗透率/束缚水列写入 DataFrame（供测井图道与交会图直接选用）。

    优先用已有核磁渗透率列（logInt 已具备其读取语义），缺失时按模型计算。
    返回新增列名列表，便于 UI 提示与报告追溯。
    """
    added = []
    if phi_col not in df.columns:
        return added

    # 束缚水：优先核磁 BVI/BVW 列，否则 Wyllie-Rose 迭代
    bv_col = next((c for c in df.columns
                   if c.upper() in ('BVI', 'BVW', 'BVM', 'SWIRR_NMR')), None)
    if bv_col and bv_col in df.columns:
        df[swirr_col] = df[bv_col].astype(float)
        added.append(swirr_col)

    if swirr_col in df.columns:
        col = f'{prefix}timur'
        df[col] = timur_perm(df[phi_col], df[swirr_col])
        added.append(col)

    ffv = next((c for c in df.columns if c.upper() in ('FFI', 'FFV', 'FFBV')), None)
    if ffv and swirr_col in df.columns:
        col = f'{prefix}coates'
        df[col] = coates_perm(df[phi_col], df[ffv], df[swirr_col])
        added.append(col)

    t2 = next((c for c in df.columns if 'T2' in c.upper() and 'LM' in c.upper()), None)
    if t2:
        col = f'{prefix}sdr'
        df[col] = sdr_perm(df[phi_col], df[t2])
        added.append(col)
    return added
```

⚠️ **必须写进界面与报告的免责提示**：这些模型的系数（`a`、`c`、`m`）是**区块相关**的，文献默认值只是初值，未经本区岩心标定直接用于产能预测会有实质误差。GUI 里要把系数做成可调并在报告中标注"未标定/已标定"状态——这是专业软件的基本严谨性，也直接关系你的解释结论可信度。

**接入点**：
- `DataProcessor` 流程末尾调用 `add_perm_columns`，新列自动进入 `results['processed_data']`
- 阶段 2 的道模板编辑器即可把 `perm_timur` 拖成新道
- 阶段 3 的交会图可加 `K-φ`（Pickett 图）组合到 `cppt.json`

**阶段 4 验收**：
1. 无核磁资料的井不产生 NaN 满屏，而是明确跳过并提示"缺少束缚水来源"
2. 有核磁资料的井，Timur / Coates / SDR 三值同深度对比图可出
